<script>

// Constants & Models

const c = 3e8; // m/s

const mu0 = 4e-7 * Math.PI;


// Simple SLW attenuation (very low for these models)

function attenFactor(media, dist, freq) {

    if (media === 'steel' || media === 'seawater') return 0.95; // near free-space for SLW

    return 1.0;

}


// Main update function

function updateSim() {

    const I = parseFloat(document.getElementById('I').value);

    const f = parseFloat(document.getElementById('f').value) * 1e6; // Hz

    const sw_km = parseFloat(document.getElementById('sw').value);

    const total_km = parseFloat(document.getElementById('d').value);

    

    document.getElementById('I_val').textContent = I.toFixed(1);

    document.getElementById('f_val').textContent = (f/1e6).toFixed(0);

    document.getElementById('sw_val').textContent = sw_km.toFixed(1);

    document.getElementById('d_val').textContent = total_km.toFixed(0);


    // Layer distances (meters)

    const d_air1 = 10;           // 33 ft inside

    const d_steel = 0.0762;      // 3 inches

    const d_sea = sw_km * 1000;

    const d_air2 = Math.max(0, total_km * 1000 - d_air1 - d_steel - d_sea);


    const r = total_km * 1000; // total range for 1/r scaling


    // Field amplitudes (simplified from Zimmerman/Hively scaling)

    const A0 = I * 1e-6; // base scaling

    const Az_zim = A0 / (r + 1);                    // ∝ 1/r

    const Ez_zim = A0 / Math.pow(r + 1, 2);        // ∝ 1/r²

    const Ez_hiv = Az_zim * 0.8;                    // Hively est.


    // Detector currents (approximate)

    const I_plasma = Az_zim * 420000 * 1e6;        // Zimmerman plasma ~420 kA/(Wb/m)

    const I_bifilar = Ez_hiv * 500;                 // Hively bifilar est.


    // SNR (example with NF)

    const txPower_dBm = 20;

    const NF = 8;

    const BW = 10000;

    const noise_dBm = -174 + 10*Math.log10(BW) + NF;

    const snr_zim = txPower_dBm - 20*Math.log10(r/1000) - noise_dBm; // rough

    const snr_hiv = snr_zim + 3; // slight advantage est.


    // Update results

    document.getElementById('results').innerHTML = `

        <strong>Total Range:</strong> ${total_km.toFixed(0)} km<br>

        <strong>Seawater Path:</strong> ${sw_km.toFixed(1)} km<br><br>

        

        <strong>Zimmerman:</strong><br>

        A_z ≈ ${Az_zim.toExponential(2)} Wb/m<br>

        E_z ≈ ${Ez_zim.toExponential(2)} V/m<br>

        Plasma Detector Current ≈ ${I_plasma.toFixed(2)} μA<br>

        SNR ≈ ${snr_zim.toFixed(1)} dB<br><br>

        

        <strong>Hively SLW:</strong><br>

        Longitudinal E ≈ ${Ez_hiv.toExponential(2)} V/m<br>

        Bifilar Coil Output ≈ ${I_bifilar.toFixed(2)} μA<br>

        SNR ≈ ${snr_hiv.toFixed(1)} dB

    `;


    // Plot (placeholder - expand with real Chart.js data)

    const ctx = document.getElementById('plot').getContext('2d');

    if (window.myChart) window.myChart.destroy();

    

    window.myChart = new Chart(ctx, {

        type: 'line',

        data: {

            labels: ['Air (33ft)', 'Steel (3")', 'Seawater', 'Space'],

            datasets: [{

                label: 'Zimmerman A_z (1/r)',

                data: [Az_zim*10, Az_zim*5, Az_zim*2, Az_zim],

                borderColor: 'blue'

            }, {

                label: 'Hively SLW E_z',

                data: [Ez_hiv*10, Ez_hiv*5, Ez_hiv*2, Ez_hiv],

                borderColor: 'red'

            }]

        },

        options: { responsive: true, scales: { y: { type: 'logarithmic' } } }

    });

}


updateSim(); // initial run
